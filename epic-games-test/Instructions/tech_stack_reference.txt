================================================================================

ASSUMED VERSION: PostgreSQL 16 (Released September 2023)

PURPOSE IN WORKFLOW:
- Player database storage
- Behavioral telemetry data
- Survey response storage (after Qualtrics export)
- Data warehouse for analysis

--------------------------------------------------------------------------------
KEY FEATURES AVAILABLE
--------------------------------------------------------------------------------

DATA TYPES:
- Standard: INTEGER, BIGINT, NUMERIC, TEXT, VARCHAR, BOOLEAN, DATE, TIMESTAMP
- JSON/JSONB: For semi-structured data (player preferences, game configs)
- ARRAY: For list data (weapons owned, achievements)

QUERY CAPABILITIES:

    - WITH clause for readable complex queries
    - Recursive CTEs for hierarchical data
    
    Aggregate Functions:
    - COUNT, SUM, AVG, MIN, MAX
    - PERCENTILE_CONT, PERCENTILE_DISC for median/percentiles
    - STRING_AGG for concatenation
    - ARRAY_AGG for collecting into arrays
    
    JSON Functions:
    - jsonb_extract_path, jsonb_array_elements
    - JSON querying with -> and ->> operators
    
    Date/Time Functions:
    - DATE_TRUNC for grouping by period
    - EXTRACT for getting date parts
    - AGE for calculating duration
    - generate_series for date ranges

PERFORMANCE FEATURES:
- Indexes (B-tree, Hash, GIN for JSON, GiST for geo)
- Table partitioning for large tables
- Parallel query execution
- Query plan analysis with EXPLAIN ANALYZE

COMMON QUERY PATTERNS FOR RESEARCH
--------------------------------------------------------------------------------

-- Audience targeting query
SELECT 
    player_id,
    email,
    region,
    platform,
    hours_played_weekly
FROM player_base
WHERE 
    last_active_date >= CURRENT_DATE - INTERVAL '30 days'
    AND account_status = 'active'
    AND email_opt_in = TRUE
    AND last_survey_date < CURRENT_DATE - INTERVAL '30 days'
ORDER BY RANDOM()
LIMIT 50000;

-- Segment distribution check
SELECT 
    player_type,
    COUNT(*) as player_count,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER(), 2) as percentage
FROM player_base
WHERE last_active_date >= CURRENT_DATE - INTERVAL '30 days'
GROUP BY player_type
ORDER BY player_count DESC;

-- Join survey to behavioral data
SELECT 
    s.response_id,
    s.player_id,
    s.sentiment_score,
    p.region,
    p.platform,
    b.hours_played_weekly,
    b.thunder_shotgun_kills,
    b.thunder_shotgun_deaths
FROM survey_responses s
LEFT JOIN player_base p ON s.player_id = p.player_id
LEFT JOIN behavioral_data b ON s.player_id = b.player_id
    AND b.date_range = 'last_30_days';

-- Cross-tab for analysis
SELECT 
    player_type,
    sentiment_label,
    COUNT(*) as response_count,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (PARTITION BY player_type), 1) as pct
FROM survey_responses s
JOIN player_base p ON s.player_id = p.player_id
GROUP BY player_type, sentiment_label
ORDER BY player_type, sentiment_label;

-- Calculate statistics for significance testing
SELECT 
    player_type,
    COUNT(*) as n,
    AVG(sentiment_score) as mean,
    STDDEV(sentiment_score) as std_dev,
    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY sentiment_score) as median
FROM survey_responses s
JOIN player_base p ON s.player_id = p.player_id
GROUP BY player_type;

-- SQL in Databricks (similar to PostgreSQL with some Delta Lake features)

-- Create survey response table with Delta Lake
CREATE TABLE IF NOT EXISTS survey_responses (
    response_id STRING,
    player_id STRING,
    survey_date DATE,
    sentiment_score INT,
    sentiment_label STRING,
    completion_time_seconds INT,
    attention_check_passed BOOLEAN
)
USING DELTA
PARTITIONED BY (survey_date);

-- Merge new survey responses (upsert)
MERGE INTO survey_responses target
USING new_responses source
ON target.response_id = source.response_id
WHEN MATCHED THEN UPDATE SET *
WHEN NOT MATCHED THEN INSERT *;

-- Time travel (view historical data)
SELECT * FROM survey_responses VERSION AS OF 5;  -- version number
SELECT * FROM survey_responses TIMESTAMP AS OF '2024-01-01';  -- point in time

# Python in Databricks

# Read survey data
df = spark.read.format("csv").option("header", "true").load("/path/to/survey.csv")

# Clean data with PySpark
from pyspark.sql.functions import col, when, avg, stddev

df_clean = df.filter(
    (col("completion_time_seconds") >= 90) &  # Remove speeders
    (col("attention_check_passed") == True)    # Remove failed attention
)

# Calculate statistics by segment
stats = df_clean.groupBy("player_type").agg(
    avg("sentiment_score").alias("mean"),
    stddev("sentiment_score").alias("std_dev"),
    count("*").alias("n")
)
stats.display()

# Run statistical test with Python
from scipy import stats
import pandas as pd

pdf = df_clean.toPandas()
casual = pdf[pdf['player_type'] == 'casual']['sentiment_score']
competitive = pdf[pdf['player_type'] == 'competitive']['sentiment_score']

t_stat, p_value = stats.ttest_ind(casual, competitive)
print(f"T-statistic: {t_stat:.3f}, P-value: {p_value:.6f}")

ATTENTION CHECK EXAMPLES:

    Type 1: Instructed Response
    "For quality purposes, please select 'Strongly Agree' for this question."
    ○ Strongly Disagree
    ○ Disagree
    ○ Neutral
    ○ Agree
    ○ Strongly Agree ← Correct
    
    Type 2: Reverse Logic
    "I have never played a video game in my entire life."
    (For a survey targeting active gamers, 'Strongly Agree' = fail)
    
    Type 3: Obvious Answer

    ○ 6

KEY CALCULATED FIELDS FOR RESEARCH
--------------------------------------------------------------------------------

-- Sentiment Score from Label
CASE [Sentiment_Label]
    WHEN 'Strongly Dislike' THEN 1
    WHEN 'Dislike' THEN 2
    WHEN 'Neutral' THEN 3
    WHEN 'Like' THEN 4
    WHEN 'Strongly Like' THEN 5
END

-- Positive/Negative Flag
IF [Sentiment_Score] >= 4 THEN 'Positive'
ELSEIF [Sentiment_Score] <= 2 THEN 'Negative'
ELSE 'Neutral'
END

-- NPS Category
IF [NPS_Score] >= 9 THEN 'Promoter'
ELSEIF [NPS_Score] <= 6 THEN 'Detractor'
ELSE 'Passive'
END

-- NPS Score (LOD calculation)
{ FIXED : 
    SUM(IF [NPS_Category] = 'Promoter' THEN 1 ELSE 0 END) / COUNT([Response_ID])
    - SUM(IF [NPS_Category] = 'Detractor' THEN 1 ELSE 0 END) / COUNT([Response_ID])
} * 100

-- Response Count with Minimum Threshold (for confidence)
IF COUNT([Response_ID]) >= 30 THEN COUNT([Response_ID])
ELSE NULL  // Don't show unreliable data
END

-- Week-over-Week Change
AVG([Sentiment_Score]) - LOOKUP(AVG([Sentiment_Score]), -1)

-- Percentage of Total
SUM([Response_Count]) / TOTAL(SUM([Response_Count]))
